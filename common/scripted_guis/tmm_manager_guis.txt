tmm_manager_tab_overview = {
    scope = character
    is_valid = {
        OR = {
            NOT = { has_variable = tmm_maa_window_open }
            NOT = { exists = var:tmm_manager_dual_recipient }
            this = var:tmm_manager_dual_recipient
        }
    }
    effect = {
        tmm_manager_open_overview_effect = yes
    }
}
