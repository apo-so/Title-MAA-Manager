# Player root. S is the native paying title; T is the row that the player clicked.
tmm_recruit_select_target = {
    scope = character
    is_valid = {
        exists = primary_title
        exists = scope:title
        scope:title = { holder = root }
        NOT = { has_variable = tmm_recruit_seen_picker }
    }
    effect = {
        tmm_province_clear_request_effect = yes
        remove_variable = tmm_recruit_success
        remove_variable = tmm_recruit_failed
        set_variable = { name = tmm_recruit_target value = scope:title }
        set_variable = { name = tmm_recruit_source value = primary_title }
        tmm_province_snapshot_effect = yes
        set_variable = { name = tmm_recruit_request_v2 value = yes }
    }
}

tmm_recruit_retarget_open = {
    scope = character
    is_valid = {
        has_variable = tmm_recruit_request_v2
        has_variable = tmm_recruit_seen_picker
        exists = scope:title
        scope:title = { holder = root }
    }
    effect = { set_variable = { name = tmm_recruit_target value = scope:title } }
}

tmm_recruit_picker_seen = {
    scope = character
    is_valid = { has_variable = tmm_recruit_request_v2 }
    effect = { set_variable = { name = tmm_recruit_seen_picker value = yes } }
}

tmm_recruit_picker_result = {
    scope = character
    is_valid = {
        has_variable = tmm_recruit_request_v2
        has_variable = tmm_recruit_seen_picker
    }
    effect = {
        tmm_recruit_resolve_effect = yes
        tmm_province_clear_request_effect = yes
    }
}
