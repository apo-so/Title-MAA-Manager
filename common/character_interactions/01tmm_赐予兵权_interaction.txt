# ===================================================
# 功能：将首都头衔兵/兵士下放给封臣的某个直属头衔（绝对流向同步版）
# 说明：通过send_option选兵种，批量或逐个将首都头衔兵转移至封臣目标头衔
# ===================================================
tmm_deploy_to_vassal_interaction = {

	# --- 基础交互设置 ---
	category = interaction_category_diplomacy
	target_type = title
	target_filter = recipient_domain_titles
	icon = request_maa_interaction
	common_interaction = no
	interface_priority = 100
	desc = tmm_deploy_to_vassal_interaction_desc
	auto_accept = yes
	

	# --- 交互是否显示 ---
	is_shown = {
		NOT = { scope:actor = scope:recipient }

		scope:recipient = {
			is_vassal_or_below_of = scope:actor
		}

		scope:actor = {
			is_ai = no
			primary_title = {
				any_owned_title_maa_regiment = {
					count >= 1
				}
			}
		}
	}
	

	# --- 目标有效性提示 ---
	has_valid_target_showing_failures_only = {
		scope:target = {
			is_leased_out = no
		}
	}


	# --- 可选择的 title 条件 ---
	can_be_picked_title = {
		scope:target = {
			is_leased_out = no
		}
	}
	

	# ===================================================
	# 选项区：根据 actor 首都兵力显示可选兵种
	# ===================================================

	# 全部头衔兵士
	send_option = {
		flag = all_title_maa
		localization = all_title_maa
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = { count >= 1 }
			}
		}
	}

	# 重步兵
	send_option = {
		flag = heavy_infantry
		localization = heavy_infantry
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = { is_unit_type = heavy_infantry }
			}
		}
	}

	# 散兵
	send_option = {
		flag = skirmishers
		localization = skirmishers
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = { is_unit_type = skirmishers }
			}
		}
	}

	# 弓箭手
	send_option = {
		flag = archers
		localization = archers
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = { is_unit_type = archers }
			}
		}
	}

	# 长矛兵
	send_option = {
		flag = pikemen
		localization = pikemen
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = { is_unit_type = pikemen }
			}
		}
	}

	# 轻骑兵
	send_option = {
		flag = light_cavalry
		localization = light_cavalry
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = { is_unit_type = light_cavalry }
			}
		}
	}

	# 重骑兵
	send_option = {
		flag = heavy_cavalry
		localization = heavy_cavalry
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = { is_unit_type = heavy_cavalry }
			}
		}
	}

	# 骆驼骑兵
	send_option = {
		flag = camel_cavalry
		localization = camel_cavalry
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = { is_unit_type = camel_cavalry }
			}
		}
	}

	# 战象骑兵
	send_option = {
		flag = elephant_cavalry
		localization = elephant_cavalry
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = { is_unit_type = elephant_cavalry }
			}
		}
	}

	# 骑射手
	send_option = {
		flag = archer_cavalry
		localization = archer_cavalry
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = { is_unit_type = archer_cavalry }
			}
		}
	}

	# 攻城武器
	send_option = {
		flag = siege_weapon
		localization = siege_weapon
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = { is_unit_type = siege_weapon }
			}
		}
	}

	# 火药部队
	send_option = {
		flag = gunpowder
		localization = gunpowder
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = { is_unit_type = gunpowder }
			}
		}
	}

	# 游牧部队
	send_option = {
		flag = nomadic_horde
		localization = nomadic_horde
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = { is_unit_type = nomadic_horde }
			}
		}
	}

	# 其他未知兵种
	send_option = {
		flag = other_type
		localization = other_type
		is_shown = {
			scope:actor.primary_title = {
				any_owned_title_maa_regiment = {
					NOR = {
						is_unit_type = skirmishers
						is_unit_type = heavy_cavalry
						is_unit_type = pikemen
						is_unit_type = heavy_infantry
						is_unit_type = siege_weapon
						is_unit_type = archers
						is_unit_type = light_cavalry
						is_unit_type = camel_cavalry
						is_unit_type = elephant_cavalry
						is_unit_type = archer_cavalry
						is_unit_type = gunpowder
						is_unit_type = nomadic_horde
					}
				}
			}
		}
	}

	# ===================================================
	# 执行区
	# ===================================================
	on_accept = {

		# --- 批量转移：全部头衔兵士（自带原生循环保护） ---
		if = {
			limit = { scope:all_title_maa = yes }
			scope:actor.primary_title = {
				every_owned_title_maa_regiment = {
					save_temporary_scope_as = old_regiment
					scope:old_regiment = { transfer_title_maa = scope:target }
					scope:actor = { tmm_transfer_maa_down = yes }
				}
			}
		}

		# --- 重步兵 ---
		if = {
			limit = {
				scope:heavy_infantry = yes
				NOT = { scope:all_title_maa = yes }
			}
			scope:actor.primary_title = {
				random_owned_title_maa_regiment = {
					limit = { is_unit_type = heavy_infantry }
					save_temporary_scope_as = old_regiment
				}
			}
			scope:actor = { tmm_transfer_maa_down = yes }
			scope:old_regiment = { transfer_title_maa = scope:target }
		}

		# --- 散兵 ---
		if = {
			limit = {
				scope:skirmishers = yes
				NOT = { scope:all_title_maa = yes }
			}
			scope:actor.primary_title = {
				random_owned_title_maa_regiment = {
					limit = { is_unit_type = skirmishers }
					save_temporary_scope_as = old_regiment
				}

			scope:actor = { tmm_transfer_maa_down = yes }
			scope:old_regiment = { transfer_title_maa = scope:target }
		}

		if = {
			limit = {
				scope:archers = yes

			}

					limit = { is_unit_type = archers }
					save_temporary_scope_as = old_regiment
				}
			}
			scope:actor = { tmm_transfer_maa_down = yes }
			scope:old_regiment = { transfer_title_maa = scope:target }
		}

		# --- 长矛兵 ---
		if = {
			limit = {
				scope:pikemen = yes
				NOT = { scope:all_title_maa = yes }
			}
			scope:actor.primary_title = {
				random_owned_title_maa_regiment = {
					limit = { is_unit_type = pikemen }
					save_temporary_scope_as = old_regiment
				}
			}
			scope:actor = { tmm_transfer_maa_down = yes }
			scope:old_regiment = { transfer_title_maa = scope:target }
		}

		# --- 轻骑兵 ---
		if = {
			limit = {
				scope:light_cavalry = yes
				NOT = { scope:all_title_maa = yes }
			}
			scope:actor.primary_title = {
				random_owned_title_maa_regiment = {
					limit = { is_unit_type = light_cavalry }
					save_temporary_scope_as = old_regiment
				}
			}
			scope:actor = { tmm_transfer_maa_down = yes }
			scope:old_regiment = { transfer_title_maa = scope:target }
		}

		# --- 重骑兵 ---
		if = {
			limit = {
				scope:heavy_cavalry = yes
				NOT = { scope:all_title_maa = yes }
			}
			scope:actor.primary_title = {
				random_owned_title_maa_regiment = {
					limit = { is_unit_type = heavy_cavalry }
					save_temporary_scope_as = old_regiment
				}
			}
			scope:actor = { tmm_transfer_maa_down = yes }
			scope:old_regiment = { transfer_title_maa = scope:target }
		}

		# --- 骆驼骑兵 ---
		if = {
			limit = {
				scope:camel_cavalry = yes
				NOT = { scope:all_title_maa = yes }
			}
			scope:actor.primary_title = {
				random_owned_title_maa_regiment = {
					limit = { is_unit_type = camel_cavalry }
					save_temporary_scope_as = old_regiment
				}
			}
			scope:actor = { tmm_transfer_maa_down = yes }
			scope:old_regiment = { transfer_title_maa = scope:target }
		}

		# --- 战象骑兵 ---
		if = {
			limit = {
				scope:elephant_cavalry = yes
				NOT = { scope:all_title_maa = yes }
			}
			scope:actor.primary_title = {
				random_owned_title_maa_regiment = {
					limit = { is_unit_type = elephant_cavalry }
					save_temporary_scope_as = old_regiment
				}
			}
			scope:actor = { tmm_transfer_maa_down = yes }
			scope:old_regiment = { transfer_title_maa = scope:target }
		}

		# --- 骑射手 ---
		if = {
			limit = {
				scope:archer_cavalry = yes
				NOT = { scope:all_title_maa = yes }
			}
			scope:actor.primary_title = {
				random_owned_title_maa_regiment = {
					limit = { is_unit_type = archer_cavalry }
					save_temporary_scope_as = old_regiment
				}
			}
			scope:actor = { tmm_transfer_maa_down = yes }
			scope:old_regiment = { transfer_title_maa = scope:target }
		}

		# --- 攻城武器 ---
		if = {
			limit = {
				scope:siege_weapon = yes
				NOT = { scope:all_title_maa = yes }
			}
			scope:actor.primary_title = {
				random_owned_title_maa_regiment = {
					limit = { is_unit_type = siege_weapon }
					save_temporary_scope_as = old_regiment
				}
			}
			scope:actor = { tmm_transfer_maa_down = yes }
			scope:old_regiment = { transfer_title_maa = scope:target }
		}

		# --- 火药部队 ---
		if = {
			limit = {
				scope:gunpowder = yes
				NOT = { scope:all_title_maa = yes }
			}
			scope:actor.primary_title = {
				random_owned_title_maa_regiment = {
					limit = { is_unit_type = gunpowder }
					save_temporary_scope_as = old_regiment
				}
			}
			scope:actor = { tmm_transfer_maa_down = yes }
			scope:old_regiment = { transfer_title_maa = scope:target }
		}

		# --- 游牧部队 ---
		if = {
			limit = {
				scope:nomadic_horde = yes
				NOT = { scope:all_title_maa = yes }
			}
			scope:actor.primary_title = {
				random_owned_title_maa_regiment = {
					limit = { is_unit_type = nomadic_horde }
					save_temporary_scope_as = old_regiment
				}
			}
			scope:actor = { tmm_transfer_maa_down = yes }
			scope:old_regiment = { transfer_title_maa = scope:target }
		}

		# --- 其他未定义兵种 ---
		if = {
			limit = {
				scope:other_type = yes
				NOT = { scope:all_title_maa = yes }
			}
			scope:actor.primary_title = {
				random_owned_title_maa_regiment = {
					limit = {
						NOR = {
							is_unit_type = skirmishers
							is_unit_type = heavy_cavalry
							is_unit_type = pikemen
							is_unit_type = heavy_infantry
							is_unit_type = siege_weapon
							is_unit_type = archers
							is_unit_type = light_cavalry
							is_unit_type = camel_cavalry
							is_unit_type = elephant_cavalry
							is_unit_type = archer_cavalry
							is_unit_type = gunpowder
							is_unit_type = nomadic_horde
						}
					}
					save_temporary_scope_as = old_regiment
				}
			}
			scope:actor = { tmm_transfer_maa_down = yes }
			scope:old_regiment = { transfer_title_maa = scope:target }
		}
	}
