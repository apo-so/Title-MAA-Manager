# ===================================================
# 功能：转移头衔兵士 V5（天下兵马）
# 说明：右键封臣 → 自动标记其所有带兵头衔 → 打开窗口 → 选头衔 → 点兵团转移
#       关闭窗口时自动清除所有标记（不留持久状态）
# ===================================================
tmm_select_title_maa_interaction = {
    category = interaction_category_diplomacy
    icon = request_maa_interaction
    common_interaction = yes
    interface_priority = 120
    desc = tmm_select_title_maa_interaction_desc
    auto_accept = yes
    ai_frequency = 0

    # --- 一、基础可用性（不限制政体类型） ---
    is_available = { always = yes }

    # --- 二、交互是否显示 ---
    is_shown = {
        scope:actor = { has_variable = tmm_reserve_interactions_open }
        NOT = { scope:actor = scope:recipient }
        scope:recipient = {
            is_vassal_or_below_of = scope:actor
        }
        scope:actor = {
            is_ai = no
        }
    }

    # --- 三、有效性（无条件放行） ---
    is_valid_showing_failures_only = {
        always = yes
    }

    # --- 四、执行：填充头衔列表 + 打开窗口 ---
    on_accept = {
        scope:recipient = {
            save_scope_as = tmm_vassal
            save_temporary_scope_as = tmm_manager_target
        }
        scope:actor = { tmm_manager_open_dual_effect = yes }
    }
}
