# ===================================================
# 功能：个人头衔兵士互转 V7（on_accept 直接填充，不经过 SGUI init）
# 说明：玩家对自己使用 → on_accept 直接填充列表和设置 flag
#       采用 find_concubine 自指模式，scope:actor = scope:recipient
#       on_accept 模仿 V5（14tmm）风格，直接操作变量列表
# ===================================================
tmm_personal_title_maa_interaction = {
    category = interaction_category_diplomacy
    icon = request_maa_interaction
    common_interaction = yes
    desc = tmm_personal_title_maa_interaction_desc
    auto_accept = yes

    # --- 一、基础可用性（不限制政体类型） ---
    is_available = { always = yes }

    # --- 二、自指模式：玩家对自己使用 ---
    is_shown = {
        scope:actor = { has_variable = tmm_reserve_interactions_open }
        scope:actor = scope:recipient
        scope:actor = {
            is_ai = no
        }
    }

    # --- 三、有效性（无条件放行） ---
    is_valid_showing_failures_only = {
        always = yes
    }

    # --- 四、按当前角色打开共同管理窗口的个人页 ---
    on_accept = {
        scope:actor = {
            tmm_manager_recover_recipient_effect = yes
            if = {
                limit = { exists = var:tmm_manager_dual_recipient NOT = { this = var:tmm_manager_dual_recipient } }
                tmm_manager_forget_dual_effect = yes
            }
            tmm_manager_open_personal_effect = yes
        }
    }
}
